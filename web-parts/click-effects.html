  const label = button.querySelector('[data-label]');
  const fallback = () => {
    const area = document.createElement('textarea'); area.value = button.dataset.value;
    area.style.position = 'fixed'; area.style.opacity = '0'; document.body.append(area); area.select();
    const copied = document.execCommand('copy');
    area.remove();
    if (!copied) throw new Error('copy failed');

  button.addEventListener('click', async () => {
    try { navigator.clipboard?.writeText ? await navigator.clipboard.writeText(button.dataset.value) : fallback(); }

      catch { label.textContent = 'コピーできませんでした'; return; }
    }
    button.classList.add('is-complete'); label.textContent = 'コピーしました';
    setTimeout(() => { button.classList.remove('is-complete'); label.textContent = 'URLをコピー'; }, 1600);
