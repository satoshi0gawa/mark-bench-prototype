HTML
<div class="c-slider" data-slider role="region" aria-roledescription="カルーセル" aria-label="おすすめ機能">
  <div class="c-slider__viewport" tabindex="0"><div class="c-slider__track">
    <article class="c-slider__slide">設計</article><article class="c-slider__slide">実装</article><article class="c-slider__slide">検証</article>

CSS
.c-slider { position: relative; width: min(100%, 20rem); }
.c-slider__viewport { overflow: hidden; border: 1px solid #18243b; }
.c-slider__track { display: flex; transition: transform .4s ease; }
.c-slider__slide { display: grid; min-width: 100%; min-height: 10rem; place-items: center; color: #fff; background: #18243b; }
.c-slider button:focus-visible, .c-slider__viewport:focus-visible { outline: 3px solid #315a9b; outline-offset: 3px; }
JAVASCRIPT
const root = document.querySelector('[data-slider]');
const track = root.querySelector('.c-slider__track');
const slides = [...track.children];
const status = root.querySelector('[data-status]');
let index = 0;
const show = (next) => {
  index = (next + slides.length) % slides.length;
  track.style.transform = `translateX(${-index * 100}%)`;
  status.textContent = `${index + 1} / ${slides.length}`;
};
root.querySelector('[data-prev]').addEventListener('click', () => show(index - 1));
root.querySelector('[data-next]').addEventListener('click', () => show(index + 1));
root.addEventListener('keydown', (event) => { if (event.key === 'ArrowLeft') show(index - 1); if (event.key === 'ArrowRight') show(index + 1); });
show(0);
USE FOR

特集、商品、導入事例などを前後ボタン・ドット・左右キーで順番に閲覧する基本形。

const root = document.querySelector('[data-auto]');
const track = root.querySelector('.c-slider__track');
const total = track.children.length;
const toggle = root.querySelector('[data-toggle]');
let index = 0, timer = 0, playing = !matchMedia('(prefers-reduced-motion: reduce)').matches;
const show = () => { track.style.transform = `translateX(${-index * 100}%)`; root.querySelector('[data-status]').textContent = `${index + 1} / ${total}`; };
const start = () => { clearInterval(timer); if (playing) timer = setInterval(() => { index = (index + 1) % total; show(); }, Number(root.dataset.interval)); };
toggle.addEventListener('click', () => { playing = !playing; toggle.textContent = playing ? '一時停止' : '再生'; start(); });
show(); start();

const root = document.querySelector('[data-drag]');
const view = root.querySelector('.c-slider__viewport');
const track = root.querySelector('.c-slider__track');
let index = 0, startX = 0, deltaX = 0;
const show = () => { track.style.transform = `translateX(${-index * 100}%)`; root.querySelector('[data-status]').textContent = `${index + 1} / ${track.children.length}`; };
view.addEventListener('pointerdown', (event) => { startX = event.clientX; view.setPointerCapture(event.pointerId); });
view.addEventListener('pointermove', (event) => { if (view.hasPointerCapture(event.pointerId)) deltaX = event.clientX - startX; });
view.addEventListener('pointerup', () => { if (Math.abs(deltaX) > view.clientWidth * .2) index = Math.max(0, Math.min(track.children.length - 1, index + (deltaX < 0 ? 1 : -1))); deltaX = 0; show(); });
show();

const root = document.querySelector('[data-loop]');
const track = root.querySelector('.c-loop__track');
const original = [...track.children];
track.prepend(original.at(-1).cloneNode(true));
track.append(original[0].cloneNode(true));
let index = 1, busy = false;
const apply = (animate = true) => { track.style.transition = animate ? '' : 'none'; track.style.transform = `translateX(${-index * 100}%)`; };
const move = (step) => { if (!busy) { busy = true; index += step; apply(); } };
root.querySelector('[data-prev]').addEventListener('click', () => move(-1));
root.querySelector('[data-next]').addEventListener('click', () => move(1));
track.addEventListener('transitionend', () => { if (index === 0) index = original.length; if (index === original.length + 1) index = 1; apply(false); requestAnimationFrame(() => { track.style.transition = ''; busy = false; }); });
apply(false);
