const root = document.querySelector('[data-hover-dropdown]');
const trigger = root.querySelector('[data-dropdown-trigger]');
const menu = root.querySelector('[data-dropdown-menu]');
const canHover = matchMedia('(hover: hover)').matches;

function setOpen(open, restoreFocus = false) {
  menu.hidden = !open;
  root.classList.toggle('is-open', open);
  trigger.setAttribute('aria-expanded', String(open));
  if (restoreFocus) trigger.focus();
}

if (canHover) {
  trigger.addEventListener('click', () => setOpen(true));
  root.addEventListener('focusin', () => setOpen(true));
  root.addEventListener('pointerenter', () => setOpen(true));
  root.addEventListener('pointerleave', () => setOpen(false));
} else {
  trigger.addEventListener('click', () => setOpen(menu.hidden));
}
root.querySelectorAll('[data-dropdown-close]').forEach((item) => {
  item.addEventListener('click', () => setOpen(false, true));
});
document.addEventListener('click', (event) => {
  if (!root.contains(event.target)) setOpen(false);
});
document.addEventListener('keydown', (event) => {
  if (event.key === 'Escape') setOpen(false, true);
});

JAVASCRIPT
(() => {
  const roots = [...document.querySelectorAll('[data-user-menu]')];

  const close = (root, restoreFocus = false) => {
    const trigger = root.querySelector('[data-user-menu-trigger]');
    const panel = root.querySelector('[data-user-menu-panel]');
    root.classList.remove('is-open');
    panel.hidden = true;
    trigger.setAttribute('aria-expanded', 'false');
    if (restoreFocus) trigger.focus();
  };

  const closeAll = (except) => {
    roots.forEach((root) => {
      if (root !== except) close(root);
    });
  };

  roots.forEach((root) => {
    const trigger = root.querySelector('[data-user-menu-trigger]');
    const panel = root.querySelector('[data-user-menu-panel]');

    const open = () => {
      closeAll(root);
      panel.hidden = false;
      root.classList.add('is-open');
      trigger.setAttribute('aria-expanded', 'true');
    };

    trigger.addEventListener('click', () => {
      panel.hidden ? open() : close(root);
    });
    trigger.addEventListener('keydown', (event) => {
      if (event.key !== 'ArrowDown') return;
      event.preventDefault();
      open();
      panel.querySelector('[role="menuitem"]')?.focus();
    });
    root.querySelectorAll('[data-user-menu-item]').forEach((item) => {
      item.addEventListener('click', () => close(root, true));
    });
  });

  document.addEventListener('click', (event) => {
    const current = event.target.closest('[data-user-menu]');
    closeAll(current);
  });
  document.addEventListener('keydown', (event) => {
    if (event.key !== 'Escape') return;
    roots.filter((root) => root.classList.contains('is-open'))
      .forEach((root) => close(root, true));
  });
})();

function setOpen(open, restoreFocus = false) {
  menu.hidden = !open;
  root.classList.toggle('is-open', open);
  trigger.setAttribute('aria-expanded', String(open));

function renderChips() {
  row.replaceChildren();
  selected.forEach((value) => {
    const chip = document.createElement('span');
    chip.className = 'c-dropdown__chip';
    chip.append(value);
    const remove = document.createElement('button');
    remove.type = 'button';
    remove.setAttribute('aria-label', `${value}を解除`);
    remove.textContent = '×';
    remove.addEventListener('click', () => {
      selected.delete(value);
      renderChips();
    });
    chip.append(remove);
    row.append(chip);
  });
  options.forEach((option) => {
    option.setAttribute('aria-pressed', String(selected.has(option.textContent.trim())));
  });
}

trigger.addEventListener('click', () => setOpen(menu.hidden));
options.forEach((option) => {
  option.addEventListener('click', () => {
    const value = option.textContent.trim();
    selected.has(value) ? selected.delete(value) : selected.add(value);
    renderChips();
  });
});

  if (!root.contains(event.target)) setOpen(false);
