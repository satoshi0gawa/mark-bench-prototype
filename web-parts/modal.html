JAVASCRIPT
const root = document.querySelector('[data-native-dialog]');
const dialog = root.querySelector('dialog');
const opener = root.querySelector('[data-dialog-open]');

const openDialog = () => {
  root.classList.add('is-open');
  dialog.show();
  dialog.querySelector('[data-dialog-close]').focus();
};
const closeDialog = () => {
  if (dialog.open) dialog.close();
};

opener.addEventListener('click', openDialog);
root.querySelectorAll('[data-dialog-close], [data-dialog-backdrop]').forEach((control) => {
  control.addEventListener('click', closeDialog);
});
root.addEventListener('keydown', (event) => {
  if (event.key === 'Escape') closeDialog();
});
dialog.addEventListener('close', () => {
  root.classList.remove('is-open');
  opener.focus();
});
USE FOR

埋め込みプレビューや管理画面の一部で、dialog要素を現在の表示領域内へ提示する場面。ページ全体を遮断する場合はshowModal()を使用します。

JAVASCRIPT
const root = document.querySelector('[data-wizard-modal]');
const dialog = root.querySelector('dialog');
const opener = root.querySelector('[data-modal-open]');
const steps = [...root.querySelectorAll('[data-wizard-step]')];
const dots = [...root.querySelectorAll('[data-wizard-dot]')];
const back = root.querySelector('[data-wizard-back]');
const next = root.querySelector('[data-wizard-next]');
let current = 0;

function render() {
  steps.forEach((step, index) => {
    step.hidden = index !== current;
    dots[index].classList.toggle('is-active', index === current);
  });
  back.disabled = current === 0;
  next.textContent = current === steps.length - 1 ? '完了' : '次へ';
}
opener.addEventListener('click', () => {
  current = 0;
  render();
  dialog.showModal();
});
back.addEventListener('click', () => {
  current = Math.max(0, current - 1);
  render();
});
next.addEventListener('click', () => {
  if (current === steps.length - 1) dialog.close();
  else { current += 1; render(); }
});
root.querySelector('[data-modal-close]')
  .addEventListener('click', () => dialog.close());
dialog.addEventListener('click', (event) => {
  if (event.target === dialog) dialog.close();
});
dialog.addEventListener('close', () => opener.focus());
render();
USE FOR

オンボーディングや初期設定。現在位置を示し、戻る・次へを一貫して配置します。

const dialog = root.querySelector('dialog');
const trigger = root.querySelector('[data-delay-open]');
const status = root.querySelector('[data-delay-status]');
const closeButton = root.querySelector('[data-modal-close]');
const key = 'campaign-modal-closed';
let timer;

function openLater() {
  window.clearTimeout(timer);
  status.textContent = '表示を予約しました';
  timer = window.setTimeout(() => {
    if (!sessionStorage.getItem(key)) dialog.showModal();
    status.textContent = '';
  }, Number(root.dataset.delay));
}
trigger.addEventListener('click', () => {
  sessionStorage.removeItem(key);
  openLater();
});
closeButton.addEventListener('click', () => {
  sessionStorage.setItem(key, '1');
  dialog.close();
});
dialog.addEventListener('click', (event) => {
  if (event.target === dialog) closeButton.click();
});
dialog.addEventListener('close', () => trigger.focus());
openLater();

function onMouseOut(event) {
  if (!event.relatedTarget && event.clientY <= 0) {
    openDialog();
    document.removeEventListener('mouseout', onMouseOut);
  }
