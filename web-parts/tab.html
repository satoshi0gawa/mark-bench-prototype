TAB

設定画面や商品詳細など、関連コンテンツを同じ領域で切り替えて表示する、Vanilla JavaScript対応のタブ12パターンです。

PATTERNS: 12TYPE: CSS + JSDEPENDENCY: NONE

BASIC

ベーシック

STANDARD
概要もっとも基本的なタブ切り替えです。
仕様ライブラリなしで複数設置できます。
価格用途に合わせて内容を配置します。
HTML
<div class="c-tab" data-basic-tabs>

    <button class="c-tab__tab" type="button" role="tab" aria-selected="false">仕様</button>
  </div>
  <div class="c-tab__panels">

JAVASCRIPT
const root = document.querySelector('[data-basic-tabs]');
const tabs = [...root.querySelectorAll('[role="tab"]')];
const panels = [...root.querySelectorAll('[role="tabpanel"]')];

function select(index, focus = false) {
  tabs.forEach((tab, i) => {
    const active = i === index;
    tab.classList.toggle('is-active', active);
    tab.setAttribute('aria-selected', String(active));
    tab.tabIndex = active ? 0 : -1;
    panels[i].hidden = !active;
  });
  if (focus) tabs[index].focus();
}
tabs.forEach((tab, index) => {
  tab.addEventListener('click', () => select(index));
  tab.addEventListener('keydown', (event) => {
    let next = null;
    if (event.key === 'ArrowRight') next = (index + 1) % tabs.length;
    if (event.key === 'ArrowLeft') next = (index - 1 + tabs.length) % tabs.length;
    if (event.key === 'Home') next = 0;
    if (event.key === 'End') next = tabs.length - 1;
    if (next === null) return;
    event.preventDefault();
    select(next, true);
  });
});

const root = document.querySelector('[data-fade-tabs]');
const tabs = [...root.querySelectorAll('[role="tab"]')];
const panels = [...root.querySelectorAll('[role="tabpanel"]')];
const reduceMotion = matchMedia('(prefers-reduced-motion: reduce)').matches;
let closeTimer = 0;

function select(index) {
  const previous = panels.findIndex((panel) => !panel.hidden);
  clearTimeout(closeTimer);
  tabs.forEach((tab, i) => {
    const active = i === index;
    tab.classList.toggle('is-active', active);
    tab.setAttribute('aria-selected', String(active));
    tab.tabIndex = active ? 0 : -1;
    panels[i].hidden = !active;
    panels[i].classList.toggle('is-active', active);
    panels[i].classList.remove('is-closing');
  });
  if (previous >= 0 && previous !== index && !reduceMotion) {
    const leaving = panels[previous];
    leaving.hidden = false;
    leaving.classList.remove('is-active');
    leaving.classList.add('is-closing');
    closeTimer = setTimeout(() => {
      leaving.classList.remove('is-closing');
      leaving.hidden = true;
    }, 200);
  }
}
tabs.forEach((tab, index) => {
  tab.addEventListener('click', () => select(index));
});
select(0);

const root = document.querySelector('[data-overflow-tabs]');
const list = root.querySelector('[role="tablist"]');
const tabs = [...root.querySelectorAll('[role="tab"]')];
const panels = [...root.querySelectorAll('[role="tabpanel"]')];
const moreWrap = root.querySelector('[data-tab-more-wrap]');
const more = root.querySelector('[data-tab-more]');
const menu = root.querySelector('[data-tab-menu]');

function select(index) {
  tabs.forEach((tab, i) => {
    const active = i === index;
    tab.classList.toggle('is-active', active);
    tab.setAttribute('aria-selected', String(active));
    panels[i].hidden = !active;
  });
}
function closeMenu() {
  menu.hidden = true;
  more.setAttribute('aria-expanded', 'false');
}
function layout() {
  tabs.forEach((tab) => { tab.hidden = false; });
  menu.replaceChildren();
  moreWrap.hidden = true;
  const total = tabs.reduce((sum, tab) => sum + tab.offsetWidth, 0);
  if (total <= list.clientWidth) return;
  moreWrap.hidden = false;
  const available = list.clientWidth - moreWrap.offsetWidth;
  let used = 0;
  tabs.forEach((tab, index) => {
    used += tab.offsetWidth;
    if (used <= available) return;
    tab.hidden = true;
    const item = document.createElement('button');
    item.type = 'button';
    item.className = 'c-tab__menu-item';
    item.textContent = tab.textContent;
    item.addEventListener('click', () => { select(index); closeMenu(); });
    menu.append(item);
  });
}
tabs.forEach((tab, index) => tab.addEventListener('click', () => select(index)));
more.addEventListener('click', () => {
  menu.hidden = !menu.hidden;
  more.setAttribute('aria-expanded', String(!menu.hidden));
});
document.addEventListener('click', (event) => {
  if (!moreWrap.contains(event.target)) closeMenu();
});
new ResizeObserver(layout).observe(list);
select(0);
layout();

const root = document.querySelector('[data-auto-tabs]');
const tabs = [...root.querySelectorAll('[role="tab"]')];
const panels = [...root.querySelectorAll('[role="tabpanel"]')];
const reduceMotion = matchMedia('(prefers-reduced-motion: reduce)').matches;
const interval = Number(root.dataset.interval) || 3000;
let current = 0;
let timer = 0;

function select(index) {
  current = index;
  tabs.forEach((tab, i) => {
    const active = i === current;
    tab.classList.toggle('is-active', active);
    tab.setAttribute('aria-selected', String(active));
    tab.tabIndex = active ? 0 : -1;
    panels[i].hidden = !active;
    panels[i].classList.toggle('is-active', active);
  });
}
const stop = () => { clearInterval(timer); timer = 0; };
const start = () => {
  if (!reduceMotion && !timer) {
    timer = setInterval(() => select((current + 1) % tabs.length), interval);
  }
};
tabs.forEach((tab, index) => {
  tab.addEventListener('click', () => select(index));
});
root.addEventListener('mouseenter', stop);
root.addEventListener('focusin', stop);
root.addEventListener('mouseleave', start);
root.addEventListener('focusout', start);
select(0);
start();
