DIRECT VALUE

HTML
<div class="c-clipboard" data-copy-text>
  <button class="c-clipboard__button" type="button"
    data-copy-trigger data-copy-value="コピーする文章">テキストをコピー</button>
  <p class="c-clipboard__feedback" data-copy-feedback aria-live="polite"></p>
</div>
CSS
.c-clipboard__button.is-done {
  color: var(--color-primary, #18243b);
  background: var(--color-accent-yellow, #ff0);
}
JAVASCRIPT
async function copyText(text) {
  try { await navigator.clipboard.writeText(text); }
  catch {
    const area = document.createElement('textarea');
    area.value = text; document.body.append(area); area.select();
    document.execCommand('copy'); area.remove();
  }
}
const root = document.querySelector('[data-copy-text]');
const button = root.querySelector('[data-copy-trigger]');
const status = root.querySelector('[data-copy-feedback]');
button.addEventListener('click', async () => {
  await copyText(button.dataset.copyValue);
  button.classList.add('is-done'); status.textContent = 'コピーしました';
  setTimeout(() => { button.classList.remove('is-done'); status.textContent = ''; }, 1600);
});
USE FOR

メールアドレス、ID、定型句など、固定文字列をワンクリックで取得させる場面。
