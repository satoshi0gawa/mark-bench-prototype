(() => {
  const reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;

const whenVisible = (element, callback) => {
    if (reduceMotion || !('IntersectionObserver' in window)) {
      callback();
      return;
    }
    let started = false;
    const observer = new IntersectionObserver((entries) => {
      if (started || !entries.some(entry => entry.isIntersecting && entry.intersectionRatio >= 0.35)) return;
      started = true;
      observer.disconnect();
      requestAnimationFrame(callback);
    }, { threshold: 0.35, rootMargin: '0px 0px -10% 0px' });
    observer.observe(element);
  };

document.querySelectorAll('[data-counter-value]').forEach((root) => {
    const output = root.querySelector('[data-counter-output]');
    const replay = root.querySelector('[data-counter-replay]');
    const target = Number(root.dataset.to || 0);
    const duration = Number(root.dataset.duration || 1200);
    const formatter = root.dataset.format === 'comma'
      ? (value) => value.toLocaleString('ja-JP')
      : (value) => String(value);
    let frame = 0;

    const play = () => {
      cancelAnimationFrame(frame);
      if (reduceMotion) {
        output.textContent = formatter(target);
        return;
      }
      const startedAt = performance.now();
      const step = (now) => {
        const progress = Math.min((now - startedAt) / duration, 1);
        const eased = 1 - Math.pow(1 - progress, 3);
        output.textContent = formatter(Math.round(target * eased));
        if (progress < 1) frame = requestAnimationFrame(step);
      };
      output.textContent = '0';
      frame = requestAnimationFrame(step);
    };

    replay?.addEventListener('click', play);
    whenVisible(root, play);
  });
})();

(() => {
  const reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;

const whenVisible = (element, callback) => {
    if (reduceMotion || !('IntersectionObserver' in window)) {
      callback();
      return;
    }
    let started = false;
    const observer = new IntersectionObserver((entries) => {
      if (started || !entries.some(entry => entry.isIntersecting && entry.intersectionRatio >= 0.35)) return;
      started = true;
      observer.disconnect();
      requestAnimationFrame(callback);
    }, { threshold: 0.35, rootMargin: '0px 0px -10% 0px' });
    observer.observe(element);
  };

document.querySelectorAll('[data-counter-roll]').forEach((root) => {
    const digits = root.querySelector('[data-counter-roll-digits]');
    const replay = root.querySelector('[data-counter-replay]');
    const columns = [];
    let timers = [];
    String(root.dataset.to || '0').split('').forEach((character) => {
      if (/\d/.test(character)) {
        const column = document.createElement('span');
        const strip = document.createElement('span');
        column.className = 'c-counter-roll__column';
        column.setAttribute('aria-hidden', 'true');
        strip.className = 'c-counter-roll__strip';
        for (let value = 0; value <= 9; value += 1) {
          const digit = document.createElement('span');
          digit.textContent = String(value);
          strip.appendChild(digit);
        }
        column.appendChild(strip);
        digits.appendChild(column);
        columns.push({ strip, value: Number(character) });
      } else {
        const separator = document.createElement('span');
        separator.className = 'c-counter-roll__separator';
        separator.setAttribute('aria-hidden', 'true');
        separator.textContent = character;
        digits.appendChild(separator);
      }
    });

    const play = () => {
      timers.forEach(window.clearTimeout);
      timers = [];
      columns.forEach(({ strip }) => {
        strip.style.transition = 'none';
        strip.style.transform = 'translateY(0)';
      });
      void digits.offsetWidth;
      columns.forEach(({ strip, value }, index) => {
        strip.style.transition = reduceMotion ? 'none' : '';
        if (reduceMotion) {
          strip.style.transform = `translateY(-${value}em)`;
          return;
        }
        timers.push(window.setTimeout(() => {
          strip.style.transform = `translateY(-${value}em)`;
        }, index * 100));
      });
    };

    replay?.addEventListener('click', play);
    whenVisible(root, play);
  });
})();

document.querySelectorAll('[data-business]').forEach((root) => {
  const toMinutes = (time) => { const [h, m] = time.split(':').map(Number); return h * 60 + m; };
  const days = root.dataset.days.split(',').map(Number);
  const update = () => {
    const now = new Date();
    const current = now.getHours() * 60 + now.getMinutes();
    const open = days.includes(now.getDay()) && current >= toMinutes(root.dataset.open) && current < toMinutes(root.dataset.close);
    root.classList.toggle('is-open', open);
    root.querySelector('[data-output]').textContent = open ? '営業中' : '営業時間外';
  };
  update(); setInterval(update, 60000);
});

document.querySelectorAll('[data-typewriter]').forEach((root) => {
  const words = root.dataset.words.split(',');
  const output = root.querySelector('[data-output]');
  let word = 0, index = 0, deleting = false;
  const type = () => {
    index += deleting ? -1 : 1;
    output.textContent = words[word].slice(0, index);
    if (!deleting && index === words[word].length) { deleting = true; setTimeout(type, 1200); return; }
    if (deleting && index === 0) { deleting = false; word = (word + 1) % words.length; }
    setTimeout(type, deleting ? 45 : 90);
  };
  type();
});

document.querySelectorAll('[data-visits]').forEach((root) => {
  const key = root.dataset.key || 'site-visits';
  const output = root.querySelector('[data-output]');
  const message = root.querySelector('[data-message]');
  const show = (count) => { output.textContent = count; message.textContent = count === 1 ? 'はじめまして！' : 'いつもありがとうございます！'; };
  let count = 1;
  try { count = Number(localStorage.getItem(key) || 0) + 1; localStorage.setItem(key, count); } catch {}
  show(count);
  root.querySelector('[data-reset]').addEventListener('click', () => {
    try { localStorage.setItem(key, '1'); } catch {}
    show(1);
  });
});
