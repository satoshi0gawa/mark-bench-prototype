document.querySelectorAll('[data-draft]').forEach((root) => {
  const field = root.querySelector('textarea');
  const bar = root.querySelector('[data-bar]');
  const status = root.querySelector('[data-status]');
  const key = root.dataset.key || 'form-draft';
  let saved = '';
  let timer;
  try { saved = localStorage.getItem(key) || ''; } catch {}
  bar.hidden = !saved;
  root.querySelector('[data-restore]').addEventListener('click', () => { field.value = saved; bar.hidden = true; });
  root.querySelector('[data-discard]').addEventListener('click', () => {
    saved = ''; field.value = ''; bar.hidden = true;
    try { localStorage.removeItem(key); } catch {}
  });
  field.addEventListener('input', () => {
    clearTimeout(timer);
    timer = setTimeout(() => {
      saved = field.value;
      try { saved ? localStorage.setItem(key, saved) : localStorage.removeItem(key); } catch {}
      bar.hidden = !saved;
      status.textContent = saved ? '下書きを自動保存しました' : '';
    }, 400);
  });
});
