apply();

  <div class="c-font__sample" data-sample>読みやすさを調整します。</div>
</div>
CSS
.c-font { width: min(100%, 24rem); font-family: sans-serif; }
.c-font button { padding: .5rem .75rem; color: #18243b; background: #fff; border: 1px solid #18243b; cursor: pointer; }

JAVASCRIPT
const root = document.querySelector('[data-font]');
const sample = root.querySelector('[data-sample]');
const buttons = [...root.querySelectorAll('[data-size]')];
const sizes = { s: '87.5%', m: '100%', l: '112.5%' };
const apply = (size) => { sample.style.setProperty('--font-size', sizes[size]); buttons.forEach((button) => button.setAttribute('aria-pressed', String(button.dataset.size === size))); try { localStorage.setItem('font-size', size); } catch {} };
buttons.forEach((button) => button.addEventListener('click', () => apply(button.dataset.size)));
let saved = 'm'; try { saved = localStorage.getItem('font-size') || 'm'; } catch {} apply(sizes[saved] ? saved : 'm');

長文記事や案内ページで、閲覧者が読みやすい文字サイズを選んで保存する場面。
