document.querySelectorAll('[data-scramble]').forEach((element) => {
  const original = element.textContent; element.setAttribute('aria-label', original); const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789#$%&'; let timer = 0;
  const play = () => {
    clearInterval(timer); if (matchMedia('(prefers-reduced-motion: reduce)').matches) return;
    let frame = 0; timer = setInterval(() => {
      const fixed = Math.floor(frame / 3);
      element.textContent = [...original].map((char, index) => char === ' ' || index < fixed ? char : chars[Math.floor(Math.random() * chars.length)]).join('');
      frame += 1; if (fixed >= original.length) { clearInterval(timer); element.textContent = original; }
    }, 38);
  };
  element.addEventListener('click', play); play();
});
