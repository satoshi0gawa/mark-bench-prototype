document.querySelectorAll('[data-accordion] .c-accordion__item').forEach((item) => {
  const button = item.querySelector('.c-accordion__button');
  const panel = item.querySelector('.c-accordion__panel');
  button.addEventListener('click', () => {
    const open = button.getAttribute('aria-expanded') !== 'true';
    item.classList.toggle('is-open', open);
    button.setAttribute('aria-expanded', String(open));
    panel.setAttribute('aria-hidden', String(!open));

});

const root = document.querySelector('[data-feature-accordion]');
const items = [...root.children];
const reduceMotion = matchMedia('(prefers-reduced-motion: reduce)');

const setLabels = (button, open) => {
  button.querySelector('[data-open-label]').hidden = open;
  button.querySelector('[data-close-label]').hidden = !open;
};

const close = async (item) => {
  const button = item.querySelector('button');
  const panel = item.querySelector('.feature-card__panel');
  if (button.getAttribute('aria-expanded') !== 'true') return;
  button.setAttribute('aria-expanded', 'false');
  setLabels(button, false);
  item.classList.remove('is-open');
  if (reduceMotion.matches) { panel.hidden = true; return; }
  panel.style.height = `${panel.scrollHeight}px`;
  panel.getBoundingClientRect();
  panel.style.transition = 'height 260ms ease';
  panel.style.height = '0px';
  await new Promise((resolve) => panel.addEventListener('transitionend', resolve, { once: true }));
  panel.hidden = true;
  panel.style.height = panel.style.transition = '';
};

const open = async (item) => {
  items.filter((other) => other !== item).forEach(close);
  const button = item.querySelector('button');
  const panel = item.querySelector('.feature-card__panel');
  button.setAttribute('aria-expanded', 'true');
  setLabels(button, true);
  item.classList.add('is-open');
  panel.hidden = false;
  if (reduceMotion.matches) return;
  panel.style.height = '0px';
  panel.style.transition = 'height 260ms ease';
  panel.getBoundingClientRect();
  panel.style.height = `${panel.scrollHeight}px`;
  await new Promise((resolve) => panel.addEventListener('transitionend', resolve, { once: true }));
  panel.style.height = 'auto';
  panel.style.transition = '';
};

items.forEach((item) => item.querySelector('button').addEventListener('click', () => {
  const openNext = item.querySelector('button').getAttribute('aria-expanded') !== 'true';
  openNext ? open(item) : close(item);
}));
